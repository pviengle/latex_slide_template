\begin{frame}{Citations}
  \TeX{} was created by Knuth \cite{knuth1984}, and \cite{lamport1994} built
  \LaTeX{} on top of it. Thai digital typography is surveyed
  in \cite{tantisiriwat2020}.

  \vspace{0.6em}
  Cite the same work twice \cite{knuth1984} and it still appears once below.
  \citequiet{mittelbach2021} and \cite*{bringhurst2012} are numbered here but
  add nothing at the bottom --- only three references are listed, not five.

  \begin{block}{What \code{\textbackslash cite} does}
    Prints \code{[n]} here, and the full reference at the bottom of this
    slide. The same work keeps the same number on the references slide at the
    end. \code{\textbackslash citequiet} gives the number only.
  \end{block}
\end{frame}

\begin{frame}{Six citations on one slide}
  Typography \cite{bringhurst2012} and typesetting \cite{knuth1984,lamport1994}
  meet document structure \cite{mittelbach2021}, script support
  \cite{unicode2024} and Thai layout \cite{tantisiriwat2020}.

  \begin{block}{When it gets crowded}
    Put \code{\textbackslash deckcitefootoff} at the top of the frame, or use
    \code{\textbackslash citequiet} for the ones that do not need a full line.
  \end{block}
\end{frame}
